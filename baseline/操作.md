
```bash
# Persistence プロジェクトの作成
$ dotnet new classlib --name Sample.Persistence --output Sample.Persistence
# ファイル作成は IDE で行う

# Domain プロジェクトの作成
$ dotnet new classlib --name Sample.Domain --output Sample.Domain
# ファイル作成は IDE で行う

# Application プロジェクトの作成
$ dotnet new classlib --name Sample.Application --output Sample.Application
# ファイル作成は IDE で行う

# Web プロジェクトの作成
$ dotnet new blazor --name Sample.Web --output Sample.Web --interactivity Server
# ファイル作成は IDE で行う

# Application から Domain への参照の追加
$ dotnet add Sample.Application/Sample.Application.csproj reference Sample.Domain/Sample.Domain.csproj

# Persistence から Application への参照の追加
$ dotnet add Sample.Persistence/Sample.Persistence.csproj reference Sample.Application/Sample.Application.csproj
# Persistence から Domain への参照の追加
$ dotnet add Sample.Persistence/Sample.Persistence.csproj reference Sample.Domain/Sample.Domain.csproj

# Web から Application への参照の追加
$ dotnet add Sample.Web/Sample.Web.csproj reference Sample.Application/Sample.Application.csproj
# Web から Persistence への参照の追加
$ dotnet add Sample.Web/Sample.Web.csproj reference Sample.Persistence/Sample.Persistence.csproj

# ソリューションの追加
$ dotnet new sln --name Sample
$ dotnet sln Sample.sln add Sample.Domain/Sample.Domain.csproj
$ dotnet sln Sample.sln add Sample.Application/Sample.Application.csproj
$ dotnet sln Sample.sln add Sample.Persistence/Sample.Persistence.csproj
$ dotnet sln Sample.sln add Sample.Web/Sample.Web.csproj
```
